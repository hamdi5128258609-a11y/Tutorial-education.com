=== FIRESTORE RULES (Firebase console > Firestore Database > Rules) ===
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isOwner() {
      return request.auth != null
        && request.auth.token.email == 'owner@example.com'   // <-- your owner email (lowercase)
        && request.auth.token.email_verified == true;
    }
    match /users/{uid} {
      allow read, write: if request.auth != null && request.auth.uid == uid
                         && request.auth.token.email != 'owner@example.com';  // <-- same email
    }
    match /videos/{id} {
      allow read: if request.auth != null;
      allow write: if isOwner();
    }
  }
}

=== STORAGE RULES (Firebase console > Storage > Rules) ===
rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    function isOwner() {
      return request.auth != null
        && request.auth.token.email == 'owner@example.com'   // <-- same email
        && request.auth.token.email_verified == true;
    }
    match /videos/{allPaths=**} {
      allow read: if request.auth != null;
      allow create: if isOwner() && request.resource.contentType.matches('video/.*');
      allow update, delete: if isOwner();
    }
  }
}
